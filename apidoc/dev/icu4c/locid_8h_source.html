1345template<typename StringClass, typename OutputIterator> inline void
1346Locale::getKeywords(OutputIterator iterator, UErrorCode& status) const
1347{
1348 if (U_FAILURE(status)) { return; }
1349 LocalPointer<StringEnumeration> keys(createKeywords(status));
1350 if (U_FAILURE(status) || keys.isNull()) {
1351 return;
1352 }
1353 for (;;) {
1354 int32_t resultLength;
1355 const char* buffer = keys->next(&resultLength, status);
1356 if (U_FAILURE(status) || buffer == nullptr) {
1357 return;
1358 }
1359 *iterator++ = StringClass(buffer, resultLength);
1360 }
1361}
