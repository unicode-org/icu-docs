180 /*
181 * C++ note:
182 * registerNode() and registerFinalValue() take ownership of their input nodes,
183 * and only return owned nodes.
184 * If they see a failure UErrorCode, they will delete the input node.
185 * If they get a nullptr pointer, they will record a U_MEMORY_ALLOCATION_ERROR.
186 * If there is a failure, they return nullptr.
187 *
188 * nullptr Node pointers can be safely passed into other Nodes because
189 * they call the static Node::hashCode() which checks for a nullptr pointer first.
190 *
191 * Therefore, as long as builder functions register a new node,
192 * they need to check for failures only before explicitly dereferencing
193 * a Node pointer, or before setting a new UErrorCode.
194 */

242 virtual int32_t markRightEdgesFirst(int32_t edgeNumber);
243 // write() must set the offset to a positive value.
244 virtual void write(StringTrieBuilder &builder) = 0;
245 // See markRightEdgesFirst.
246 inline void writeUnlessInsideRightEdge(int32_t firstRight, int32_t lastRight,
247 StringTrieBuilder &builder) {
248 // Note: Edge numbers are negative, lastRight<=firstRight.
249 // If offset>0 then this node and its sub-nodes have been written already
250 // and we need not write them again.
251 // If this node is part of the unwritten right branch edge,
252 // then we wait until that is written.
253 if(offset<0 && (offset<lastRight || firstRight<offset)) {
254 write(builder);
255 }
256 }
257 inline int32_t getOffset() const { return offset; }
258 protected:
259 int32_t hash;
260 int32_t offset;
