697 /*
698 * This is for ICU internal use only. Please do not use.
699 * Save the interval pattern information.
700 * Interval pattern consists of 2 single date patterns and the separator.
701 * For example, interval pattern "MMM d - MMM d, yyyy" consists
702 * a single date pattern "MMM d", another single date pattern "MMM d, yyyy",
703 * and a separator "-".
704 * The pattern is divided into 2 parts. For above example,
705 * the first part is "MMM d - ", and the second part is "MMM d, yyyy".
706 * Also, the first date appears in an interval pattern could be
707 * the earlier date or the later date.
708 * And such information is saved in the interval pattern as well.
709 */
710 struct PatternInfo {
711 UnicodeString firstPart;
712 UnicodeString secondPart;
726 UBool laterDateFirst;
727 };
