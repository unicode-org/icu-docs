750 /*
751 * Templatized version of handleNext() and handleSafePrevious().
752 *
753 * There will be exactly four instantiations, two each for 8 and 16 bit tables,
754 * two each for 8 and 16 bit trie.
755 * Having separate instantiations for the table types keeps conditional tests of
756 * the table type out of the inner loops, at the expense of replicated code.
757 *
758 * The template parameter for the Trie access function is a value, not a type.
759 * Doing it this way, the compiler will inline the Trie function in the
760 * expanded functions. (Both the 8 and 16 bit access functions have the same type
761 * signature)
762 */
763
764 typedef uint16_t (*PTrieFunc)(const UCPTrie *, UChar32);
765
766 template<typename RowType, PTrieFunc trieFunc>
767 int32_t handleSafePrevious(int32_t fromPosition);
768
769 template<typename RowType, PTrieFunc trieFunc>
