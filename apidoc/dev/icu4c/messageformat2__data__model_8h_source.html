35// Note: this _must_ be declared `inline` or else gcc will generate code
36// for its instantiations, which needs to be avoided because it returns
37// a std::vector
38template<typename T>
39static inline std::vector<T> toStdVector(const T* arr, int32_t len) {
40 std::vector<T> result;
41 for (int32_t i = 0; i < len; i++) {
42 result.push_back(arr[i]);
43 }
44 return result;
45}

1358 /*
1359 Internally, an expression is represented as the application of an optional operator to an operand.
1360 The operand is always present; for function calls with no operand, it's represented
1361 as an operand for which `isNull()` is true.
1362
1363 Operator | Operand
1364 --------------------------------
1365 { |42| :fun opt=value } => (FunctionName=fun, | Literal(quoted=true, contents="42")
1366 options={opt: value})
1367 { abcd } => null | Literal(quoted=false, contents="abcd")
1368 { : fun opt=value } => (FunctionName=fun,
1369 options={opt: value}) | NullOperand()
1370 */
1371
1372 Expression(const Operator &rAtor, const Operand &rAnd, const OptionMap& attrs) : rator(rAtor), rand(rAnd), attributes(attrs) {}
1373 Expression(const Operand &rAnd, const OptionMap& attrs) : rator(std::nullopt), rand(Operand(rAnd)), attributes(attrs) {}
1374 Expression(const Operator &rAtor, const OptionMap& attrs) : rator(rAtor), rand(), attributes(attrs) {}
1375 /* const */ std::optional<Operator> rator;
1376 /* const */ Operand rand;
1377 /* const */ OptionMap attributes;

2097 class U_I18N_API_CLASS MFDataModel : public UMemory {
2098 /*
2099 Classes that represent nodes in the data model are nested inside the
2100 `MFDataModel` class.
2101
2102 Classes such as `Expression`, `Pattern` and `VariantMap` are immutable and
2103 are constructed using the builder pattern.
2104
2105 Most classes representing nodes have copy constructors. This is because builders
2106 contain immutable data that must be copied when calling `build()`, since the builder
2107 could go out of scope before the immutable result of the builder does. Copying is
2108 also necessary to prevent unexpected mutation if intermediate builders are saved
2109 and mutated again after calling `build()`.
2110
2111 The copy constructors perform a deep copy, for example by copying the entire
2112 list of options for an `Operator` (and copying the entire underlying vector.)
2113 Some internal fields should be `const`, but are declared as non-`const` to make
2114 the copy constructor simpler to implement. (These are noted throughout.) In
2115 other words, those fields are `const` except during the execution of a copy
2116 constructor.
2117
2118 On the other hand, intermediate `Builder` methods that return a `Builder&`
2119 mutate the state of the builder, so in code like:
2120
2121 Expression::Builder& exprBuilder = Expression::builder()-> setOperand(foo);
2122 Expression::Builder& exprBuilder2 = exprBuilder.setOperator(bar);
2123
2124 the call to `setOperator()` would mutate `exprBuilder`, since `exprBuilder`
2125 and `exprBuilder2` are references to the same object.
2126
2127 An alternate choice would be to make `build()` destructive, so that copying would
2128 be unnecessary. Or, both copying and moving variants of `build()` could be
2129 provided. Copying variants of the intermediate `Builder` methods could be
2130 provided as well, if this proved useful.
2131 */
