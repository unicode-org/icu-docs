1336public:
1338 USetCodePointIterator(const USetCodePointIterator &other) = default;
1339
1341 bool operator==(const USetCodePointIterator &other) const {
1342 // No need to compare rangeCount & end given private constructor
1343 // and assuming we don't compare iterators across the set being modified.
1344 // And comparing rangeIndex is redundant with comparing c.
1345 // We might even skip comparing uset.
1346 // Unless we want operator==() to be "correct" for more than iteration.
1347 return uset == other.uset && c == other.c;
1348 }
1349
1351 bool operator!=(const USetCodePointIterator &other) const { return !operator==(other); }
1352
1354 UChar32 operator*() const { return c; }
1355
1360 USetCodePointIterator& operator++() U_LIFETIME_BOUND {
1361 if (c < end) {
1362 ++c;
1363 } else if (rangeIndex < rangeCount) {
1364 UErrorCode errorCode = U_ZERO_ERROR;
1365 int32_t result = uset_getItem(uset, rangeIndex, &c, &end, nullptr, 0, &errorCode);
1366 if (U_SUCCESS(errorCode) && result == 0) {
1367 ++rangeIndex;
1368 } else {
1369 c = end = U_SENTINEL;
1370 }
1371 } else {
1372 c = end = U_SENTINEL;
1373 }
1374 return *this;
1375 }

1393 : uset(pUset), rangeIndex(nRangeIndex), rangeCount(nRangeCount),
