389#define UCPTRIE_FAST_U16_NEXT(trie, dataAccess, src, limit, c, result) UPRV_BLOCK_MACRO_BEGIN { \
390 (c) = *(src)++; \
391 int32_t __index; \
392 if (!U16_IS_SURROGATE(c)) { \
393 __index = _UCPTRIE_FAST_INDEX(trie, c); \
394 } else { \
395 uint16_t __c2; \
396 if (U16_IS_SURROGATE_LEAD(c) && (src) != (limit) && U16_IS_TRAIL(__c2 = *(src))) { \
397 ++(src); \
398 (c) = U16_GET_SUPPLEMENTARY((c), __c2); \
399 __index = _UCPTRIE_SMALL_INDEX(trie, c); \
400 } else { \
401 __index = (trie)->dataLength - UCPTRIE_ERROR_VALUE_NEG_DATA_OFFSET; \
402 } \
403 } \
404 (result) = dataAccess(trie, __index); \
405} UPRV_BLOCK_MACRO_END

420#define UCPTRIE_FAST_U16_PREV(trie, dataAccess, start, src, c, result) UPRV_BLOCK_MACRO_BEGIN { \
421 (c) = *--(src); \
422 int32_t __index; \
423 if (!U16_IS_SURROGATE(c)) { \
424 __index = _UCPTRIE_FAST_INDEX(trie, c); \
425 } else { \
426 uint16_t __c2; \
427 if (U16_IS_SURROGATE_TRAIL(c) && (src) != (start) && U16_IS_LEAD(__c2 = *((src) - 1))) { \
428 --(src); \
429 (c) = U16_GET_SUPPLEMENTARY(__c2, (c)); \
430 __index = _UCPTRIE_SMALL_INDEX(trie, c); \
431 } else { \
432 __index = (trie)->dataLength - UCPTRIE_ERROR_VALUE_NEG_DATA_OFFSET; \
433 } \
434 } \
435 (result) = dataAccess(trie, __index); \
436} UPRV_BLOCK_MACRO_END

454#define UCPTRIE_FAST_U8_NEXT(trie, dataAccess, src, limit, result) UPRV_BLOCK_MACRO_BEGIN { \
455 int32_t __lead = (uint8_t)*(src)++; \
456 if (!U8_IS_SINGLE(__lead)) { \
457 uint8_t __t1, __t2, __t3; \
458 if ((src) != (limit) && \
459 (__lead >= 0xe0 ? \
460 __lead < 0xf0 ? /* U+0800..U+FFFF except surrogates */ \
461 U8_LEAD3_T1_BITS[__lead &= 0xf] & (1 << ((__t1 = *(src)) >> 5)) && \
462 ++(src) != (limit) && (__t2 = *(src) - 0x80) <= 0x3f && \
463 (__lead = ((int32_t)(trie)->index[(__lead << 6) + (__t1 & 0x3f)]) + __t2, 1) \
464 : /* U+10000..U+10FFFF */ \
465 (__lead -= 0xf0) <= 4 && \
466 U8_LEAD4_T1_BITS[(__t1 = *(src)) >> 4] & (1 << __lead) && \
467 (__lead = (__lead << 6) | (__t1 & 0x3f), ++(src) != (limit)) && \
468 (__t2 = *(src) - 0x80) <= 0x3f && \
469 ++(src) != (limit) && (__t3 = *(src) - 0x80) <= 0x3f && \
470 (__lead = __lead >= (trie)->shifted12HighStart ? \
471 (trie)->dataLength - UCPTRIE_HIGH_VALUE_NEG_DATA_OFFSET : \
472 ucptrie_internalSmallU8Index((trie), __lead, __t2, __t3), 1) \
473 : /* U+0080..U+07FF */ \
474 __lead >= 0xc2 && (__t1 = *(src) - 0x80) <= 0x3f && \
475 (__lead = (int32_t)(trie)->index[__lead & 0x1f] + __t1, 1))) { \
476 ++(src); \
477 } else { \
478 __lead = (trie)->dataLength - UCPTRIE_ERROR_VALUE_NEG_DATA_OFFSET; /* ill-formed*/ \
479 } \
480 } \
481 (result) = dataAccess(trie, __lead); \
482} UPRV_BLOCK_MACRO_END

500#define UCPTRIE_FAST_U8_PREV(trie, dataAccess, start, src, result) UPRV_BLOCK_MACRO_BEGIN { \
501 int32_t __index = (uint8_t)*--(src); \
502 if (!U8_IS_SINGLE(__index)) { \
503 __index = ucptrie_internalU8PrevIndex((trie), __index, (const uint8_t *)(start), \
504 (const uint8_t *)(src)); \
505 (src) -= __index & 7; \
506 __index >>= 3; \
507 } \
508 (result) = dataAccess(trie, __index); \
509} UPRV_BLOCK_MACRO_END
