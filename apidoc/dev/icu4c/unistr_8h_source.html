4090 /*
4091 * The following are all the class fields that are stored
4092 * in each UnicodeString object.
4093 * Note that UnicodeString has virtual functions,
4094 * therefore there is an implicit vtable pointer
4095 * as the first real field.
4096 * The fields should be aligned such that no padding is necessary.
4097 * On 32-bit machines, the size should be 32 bytes,
4098 * on 64-bit machines (8-byte pointers), it should be 40 bytes.
4099 *
4100 * We use a hack to achieve this.
4101 *
4102 * With at least some compilers, each of the following is forced to
4103 * a multiple of sizeof(pointer) [the largest field base unit here is a data pointer],
4104 * rounded up with additional padding if the fields do not already fit that requirement:
4105 * - sizeof(class UnicodeString)
4106 * - offsetof(UnicodeString, fUnion)
4107 * - sizeof(fUnion)
4108 * - sizeof(fStackFields)
4109 *
4110 * We optimize for the longest possible internal buffer for short strings.
4111 * fUnion.fStackFields begins with 2 bytes for storage flags
4112 * and the length of relatively short strings,
4113 * followed by the buffer for short string contents.
4114 * There is no padding inside fStackFields.
4115 *
4116 * Heap-allocated and aliased strings use fUnion.fFields.
4117 * Both fStackFields and fFields must begin with the same fields for flags and short length,
4118 * that is, those must have the same memory offsets inside the object,
4119 * because the flags must be inspected in order to decide which half of fUnion is being used.
4120 * We assume that the compiler does not reorder the fields.
4121 *
4122 * (Padding at the end of fFields is ok:
4123 * As long as it is no larger than fStackFields, it is not wasted space.)
4124 *
4125 * For some of the history of the UnicodeString class fields layout, see
4126 * - ICU ticket #11551 "longer UnicodeString contents in stack buffer"
4127 * - ICU ticket #11336 "UnicodeString: recombine stack buffer arrays"
4128 * - ICU ticket #8322 "why is sizeof(UnicodeString)==48?"
4129 */
4130 // (implicit) *vtable;
4131 union StackBufferOrFields {
4132 // fStackFields is used iff (fLengthAndFlags&kUsingStackBuffer) else fFields is used.
4133 // Each struct of the union must begin with fLengthAndFlags.
4134 struct {
4135 int16_t fLengthAndFlags; // bit fields: see constants above
4136 char16_t fBuffer[US_STACKBUF_SIZE]; // buffer for short strings
4137 } fStackFields;
4138 struct {
4139 int16_t fLengthAndFlags; // bit fields: see constants above
4140 int32_t fLength; // number of characters in fArray if >127; else undefined
4141 int32_t fCapacity; // capacity of fArray (in char16_ts)
4142 // array pointer last to minimize padding for machines with P128 data model
4143 // or pointer sizes that are not a power of 2
4144 char16_t *fArray; // the Unicode data
4145 } fFields;
4146 } fUnion;
4147};
